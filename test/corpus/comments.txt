==================

/// Model doc.

  /// Field doc.
  #1 x int32
}

/// Enum doc.

  /// Value doc.
  #1 v
}

/// Union doc.
union U =
  /// First member.
  #1 A
  | /// Second member.
  #2 B

/// Alias doc.
alias M = int32
