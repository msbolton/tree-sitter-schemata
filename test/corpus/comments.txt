==================
doc comments attach to elements
==================

/// Namespace doc.
namespace a

/// Record doc.
record R {
  /// Field doc.
  #1 x: int32
}

/// Enum doc.
enum E {
  /// Value doc.
  #1 v
}

/// Union doc.
union U =
  /// First member.
  #1 A
  | /// Second member.
  #2 B

/// Alias doc.
alias M = int32

---

(source_file
  (doc_comment)
  (namespace_declaration name: (qualified_name (identifier)))
  (record_declaration
    (doc_comment)
    name: (identifier)
    body: (record_body
      (field
        (doc_comment)
        ordinal: (ordinal)
        name: (identifier)
        type: (type name: (qualified_name (identifier))))))
  (enum_declaration
    (doc_comment)
    name: (identifier)
    body: (enum_body
      (enum_value
        (doc_comment)
        ordinal: (ordinal)
        name: (identifier))))
  (union_declaration
    (doc_comment)
    name: (identifier)
    (union_member
      (doc_comment)
      ordinal: (ordinal)
      type: (type name: (qualified_name (identifier))))
    (union_member
      (doc_comment)
      ordinal: (ordinal)
      type: (type name: (qualified_name (identifier)))))
  (alias_declaration
